  **docker pull** - установка образа контейнера.
```bash
docker pull <имя образа>:<тэг> # можно не юзать тег тогда по умолчанию будет последний тэг
```

**docker search** - поиск образа
```bash
docker search <иия образа>
```

**docker run** - запуск образа контейнера создает контейнер
```bash
docker run <имя образа> # Просто запустит контейнер

docker run --name <имя контейнера> <имя образа> # создаст контейнер и сразу запустит его 

docker run -i -t <имя образа> # зайти внутрь контейнера 
```

**docker ps** - проверить контейнеры
```bash
docker ps -a # проверяет все контейнеры
docker ps # проверяет только запущенные контейнеры
```

**docker start** - запустить уже запущенный когда то контейнер
```bash
docker start <имя контейнера> # запустить просто

docker start -i # Покажет вывод запущенного контейнера>
```

**docker stop** - остановить рабочий контейнер
```bash
docker stop <имя контейнера>
```

**docker kill** - принудительно остановить контейнер
```bash
docker kill <имя контейнера>
```

**docker rm** - удалить контейнер
```bash
docker rm <имя контейнера> # удаляем по имени контейнера

docker rm <айди контейнера> # удаляем по айди контейнера

docker container prune # удаляет все остановленные контейнеры

docker rmi <имя образа>:<тэг> # удалить образ контейнера. Если не указан тэг то удалиться последний контейнер   
```

**docker rmi** - удалить образ
```bash
docker rmi
```

**docker images** - глянуть образы
```bash
docker images # показать все образы

docker images -a # показать что то скрытое хз
```

**docker history** - глянуть слои из которых состоит образ
```bash
docker history <имя образа> # смотрит все слои
```
